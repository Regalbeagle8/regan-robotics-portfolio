function Button({ children, className = "", variant = "default", ...props }) {
  const base =
    "inline-flex items-center justify-center rounded-full font-semibold transition focus:outline-none focus:ring-2 focus:ring-sky-300 focus:ring-offset-2 focus:ring-offset-slate-950 disabled:pointer-events-none disabled:opacity-50";
  const styles =
    variant === "outline"
      ? "border border-white/15 bg-white/5 text-white hover:bg-white/10 hover:text-sky-200"
      : "bg-sky-400 text-slate-950 hover:bg-sky-300";
  return (
    <button className={`${base} ${styles} ${className}`} {...props}>
      {children}
    </button>
  );
}

function Card({ children, className = "" }) {
  return <div className={`rounded-2xl ${className}`}>{children}</div>;
}

function CardContent({ children, className = "" }) {
  return <div className={className}>{children}</div>;
}

function scrollToSection(id) {
  const target = document.getElementById(id);
  if (target) target.scrollIntoView({ behavior: "smooth", block: "start" });
}

function SectionHeading({ eyebrow, title, children }) {
  return (
    <div className="mx-auto mb-12 max-w-3xl text-center">
      <p className="mb-3 text-sm font-semibold uppercase tracking-[0.24em] text-sky-400">
        {eyebrow}
      </p>
      <h2 className="text-3xl font-bold tracking-tight text-white md:text-5xl">{title}</h2>
      {children && <p className="mt-5 text-base leading-8 text-slate-300 md:text-lg">{children}</p>}
    </div>
  );
}

      setShowTop(window.scrollY > 550);
      const current = sections
        .map((section) => {
          const el = document.getElementById(section.id);
          if (!el) return { id: section.id, distance: Infinity };
          return { id: section.id, distance: Math.abs(el.getBoundingClientRect().top - 120) };
